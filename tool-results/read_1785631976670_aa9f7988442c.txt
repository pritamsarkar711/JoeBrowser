   196→   196→function buildLaunchArguments(profile: ProfileData, profileDir: string, extensionDir: string): string[] {
   197→   197→  const isFirefox = profile.browserType === 'firefox';
   198→   198→  const args: string[] = [];
   199→   199→
   200→   200→  if (isFirefox) {
   201→   201→    // Firefox-specific arguments
   202→   202→    args.push('-profile', profileDir);
   203→   203→    args.push('-no-remote'); // Don't reuse existing Firefox instance
   204→   204→    args.push('-new-instance');
   205→   205→
   206→   206→    // Set launch URL
   207→   207→    if (profile.launchUrl) {
   208→   208→      args.push(profile.launchUrl);
   209→   209→    }
   210→   210→
   211→   211→    // Proxy configuration for Firefox
   212→   212→    if (profile.proxy) {
   213→   213→      // Firefox proxy is set via user.js in the profile directory
   214→   214→      createFirefoxProxyConfig(profileDir, profile.proxy);
   215→   215→    }
   216→   216→  } else {
   217→   217→    // Chrome/Chromium/Brave/Edge arguments
   218→   218→    args.push(`--user-data-dir=${profileDir}`);
   219→   219→    args.push(`--load-extension=${extensionDir}`);
   220→   220→    args.push('--no-first-run');
   221→   221→    args.push('--no-default-browser-check');
   222→   222→    args.push('--disable-background-networking');
   223→   223→    args.push('--disable-client-side-phishing-detection');
   224→   224→    args.push('--disable-default-apps');
   225→   225→    args.push('--disable-hang-monitor');
   226→   226→    args.push('--disable-popup-blocking');
   227→   227→    args.push('--disable-prompt-on-repost');
   228→   228→    args.push('--disable-sync');
   229→   229→    args.push('--disable-web-security');
   230→   230→    args.push('--metrics-recording-only');
   231→   231→    args.push('--safebrowsing-disable-auto-update');
   232→   232→
   233→   233→    // Set user agent
   234→   234→    if (profile.fingerprint.userAgent) {
   235→   235→      args.push(`--user-agent=${profile.fingerprint.userAgent}`);
   236→   236→    }
   237→   237→
   238→   238→    // Set window size for mobile emulation
   239→   239→    if (profile.deviceType === 'mobile') {
   240→   240→      const screenRes = profile.fingerprint.screenResolution.split('x');
   241→   241→      args.push(`--window-size=${screenRes[0]},${screenRes[1]}`);
   242→   242→    }
   243→   243→
   244→   244→    // Proxy configuration
   245→   245→    if (profile.proxy) {
   246→   246→      const proxyUrl = `${profile.proxy.type}://${profile.proxy.host}:${profile.proxy.port}`;
   247→   247→      args.push(`--proxy-server=${proxyUrl}`);
   248→   248→
   249→   249→      // Proxy authentication via extension
   250→   250→      if (profile.proxy.username && profile.proxy.password) {
   251→   251→        createProxyAuthExtension(profileDir, profile.proxy);
   252→   252→      }
   253→   253→    }
   254→   254→
   255→   255→    // Launch URL
   256→   256→    if (profile.launchUrl) {
   257→   257→      args.push(profile.launchUrl);
   258→   258→    }
   259→   259→  }
   260→   260→
   261→   261→  return args;
   262→   262→}

   315→   315→function createFirefoxProxyConfig(profileDir: string, proxy: ProfileData['proxy']): void {
   316→   316→  if (!proxy) return;
   317→   317→
   318→   318→  const prefsPath = path.join(profileDir, 'user.js');
   319→   319→  const existing = fs.existsSync(prefsPath) ? fs.readFileSync(prefsPath, 'utf8') : '';
   320→   320→
   321→   321→  let proxyPrefs = existing + '\n// Proxy Configuration\n';
   322→   322→
   323→   323→  switch (proxy.type) {
   324→   324→    case 'http':
   325→   325→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   326→   326→      proxyPrefs += `user_pref("network.proxy.http", "${proxy.host}");\n`;
   327→   327→      proxyPrefs += `user_pref("network.proxy.http_port", ${proxy.port});\n`;
   328→   328→      proxyPrefs += `user_pref("network.proxy.ssl", "${proxy.host}");\n`;
   329→   329→      proxyPrefs += `user_pref("network.proxy.ssl_port", ${proxy.port});\n`;
   330→   330→      break;
   331→   331→    case 'socks4':
   332→   332→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   333→   333→      proxyPrefs += `user_pref("network.proxy.socks", "${proxy.host}");\n`;
   334→   334→      proxyPrefs += `user_pref("network.proxy.socks_port", ${proxy.port});\n`;
   335→   335→      proxyPrefs += `user_pref("network.proxy.socks_version", 4);\n`;
   336→   336→      break;
   337→   337→    case 'socks5':
   338→   338→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   339→   339→      proxyPrefs += `user_pref("network.proxy.socks", "${proxy.host}");\n`;
   340→   340→      proxyPrefs += `user_pref("network.proxy.socks_port", ${proxy.port});\n`;
   341→   341→      proxyPrefs += `user_pref("network.proxy.socks_version", 5);\n`;
   342→   342→      break;
   343→   343→    case 'https':
   344→   344→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   345→   345→      proxyPrefs += `user_pref("network.proxy.http", "${proxy.host}");\n`;
   346→   346→      proxyPrefs += `user_pref("network.proxy.http_port", ${proxy.port});\n`;
   347→   347→      proxyPrefs += `user_pref("network.proxy.ssl", "${proxy.host}");\n`;
   348→   348→      proxyPrefs += `user_pref("network.proxy.ssl_port", ${proxy.port});\n`;
   349→   349→      break;
   350→   350→  }
   351→   351→
   352→   352→  if (proxy.username) {
   353→   353→    proxyPrefs += `user_pref("network.proxy.share_proxy_settings", true);\n`;
   354→   354→    // Firefox doesn't support proxy auth in prefs directly — need extension
   355→   355→    // We'll handle this via the stealth extension
   356→   356→  }
   357→   357→
   358→   358→  proxyPrefs += `user_pref("network.proxy.no_proxies_on", "localhost, 127.0.0.1");\n`;
   359→   359→
   360→   360→  fs.writeFileSync(prefsPath, proxyPrefs, 'utf8');
   361→   361→}

   399→   399→function copyExtensionFiles(sourceDir: string, targetDir: string): boolean {
   400→   400→  try {
   401→   401→    if (!fs.existsSync(sourceDir)) return false;
   402→   402→
   403→   403→    const files = fs.readdirSync(sourceDir);
   404→   404→    for (const file of files) {
   405→   405→      const src = path.join(sourceDir, file);
   406→   406→      const dst = path.join(targetDir, file);
   407→   407→      fs.copyFileSync(src, dst);
   408→   408→    }
   409→   409→    return true;
   410→   410→  } catch (err) {
   411→   411→    return false;
   412→   412→  }
