    32→/**
    33→ * Close a running profile's browser
    34→ */
    35→export function closeProfileBrowser(profileId: string): boolean {
    36→  const proc = runningProcesses.get(profileId);
    37→  if (proc && !proc.killed) {
    38→    proc.kill();
    39→    runningProcesses.delete(profileId);
    40→    return true;
    41→  }
    42→  runningProcesses.delete(profileId);
    43→  return false;
    44→}

    46→/**
    47→ * Launch a profile in a REAL browser
    48→ */
    49→export async function launchProfile(profile: ProfileData): Promise<{ success: boolean; error?: string }> {
    50→  try {
    51→    // Check if already running
    52→    if (runningProcesses.has(profile.id)) {
    53→      return { success: true };
    54→    }
    55→
    56→    if (!profile.id || !profile.fingerprint) {
    57→      return { success: false, error: 'Invalid profile data' };
    58→    }
    59→
    60→    // Auto-detect timezone from proxy if proxy is set
    61→    if (profile.proxy) {
    62→      await autoDetectTimezoneFromProxy(profile);
    63→    }
    64→
    65→    // Create profile directory
    66→    const profileDir = createProfileDirectory(profile);
    67→    if (!profileDir) {
    68→      return { success: false, error: 'Failed to create profile directory' };
    69→    }
    70→
    71→    // Create stealth extension for this profile
    72→    const extensionDir = createStealthExtension(profile);
    73→    if (!extensionDir) {
    74→      return { success: false, error: 'Failed to create stealth extension' };
    75→    }
    76→
    77→    // Find the browser executable
    78→    const browserPath = findBrowserExecutable(profile.browserType);
    79→    if (!browserPath) {
    80→      return { success: false, error: `${getBrowserName(profile.browserType)} not found. Please install it first.` };
    81→    }
    82→
    83→    // Build launch arguments
    84→    const args = buildLaunchArguments(profile, profileDir, extensionDir);
    85→
    86→    // Launch the browser
    87→    const proc = spawn(browserPath, args, {
    88→      detached: false,
    89→      stdio: 'ignore',
    90→    });
    91→
    92→    if (!proc.pid) {
    93→      return { success: false, error: 'Failed to start browser process' };
    94→    }
    95→
    96→    // Track the process
    97→    runningProcesses.set(profile.id, proc);
    98→
    99→    // Handle process exit
   100→    proc.on('exit', () => {
   101→      runningProcesses.delete(profile.id);
   102→    });
   103→
   104→    proc.on('error', (err) => {
   105→      console.error(`[RealBrowserLauncher] Process error for ${profile.id}:`, err);
   106→      runningProcesses.delete(profile.id);
   107→    });
   108→
   109→    return { success: true };
   110→
   111→  } catch (err: any) {
   112→    console.error('[RealBrowserLauncher] Failed to launch profile:', err);
   113→    return { success: false, error: err.message || 'Unknown error launching profile' };
   114→  }
   115→}

   196→function buildLaunchArguments(profile: ProfileData, profileDir: string, extensionDir: string): string[] {
   197→  const isFirefox = profile.browserType === 'firefox';
   198→  const args: string[] = [];
   199→
   200→  if (isFirefox) {
   201→    // Firefox-specific arguments
   202→    args.push('-profile', profileDir);
   203→    args.push('-no-remote'); // Don't reuse existing Firefox instance
   204→    args.push('-new-instance');
   205→
   206→    // Set launch URL
   207→    if (profile.launchUrl) {
   208→      args.push(profile.launchUrl);
   209→    }
   210→
   211→    // Proxy configuration for Firefox
   212→    if (profile.proxy) {
   213→      // Firefox proxy is set via user.js in the profile directory
   214→      createFirefoxProxyConfig(profileDir, profile.proxy);
   215→    }
   216→  } else {
   217→    // Chrome/Chromium/Brave/Edge arguments
   218→    args.push(`--user-data-dir=${profileDir}`);
   219→    args.push(`--load-extension=${extensionDir}`);
   220→    args.push('--no-first-run');
   221→    args.push('--no-default-browser-check');
   222→    args.push('--disable-background-networking');
   223→    args.push('--disable-client-side-phishing-detection');
   224→    args.push('--disable-default-apps');
   225→    args.push('--disable-hang-monitor');
   226→    args.push('--disable-popup-blocking');
   227→    args.push('--disable-prompt-on-repost');
   228→    args.push('--disable-sync');
   229→    args.push('--disable-web-security');
   230→    args.push('--metrics-recording-only');
   231→    args.push('--safebrowsing-disable-auto-update');
   232→
   233→    // Set user agent
   234→    if (profile.fingerprint.userAgent) {
   235→      args.push(`--user-agent=${profile.fingerprint.userAgent}`);
   236→    }
   237→
   238→    // Set window size for mobile emulation
   239→    if (profile.deviceType === 'mobile') {
   240→      const screenRes = profile.fingerprint.screenResolution.split('x');
   241→      args.push(`--window-size=${screenRes[0]},${screenRes[1]}`);
   242→    }
   243→
   244→    // Proxy configuration
   245→    if (profile.proxy) {
   246→      const proxyUrl = `${profile.proxy.type}://${profile.proxy.host}:${profile.proxy.port}`;
   247→      args.push(`--proxy-server=${proxyUrl}`);
   248→
   249→      // Proxy authentication via extension
   250→      if (profile.proxy.username && profile.proxy.password) {
   251→        createProxyAuthExtension(profileDir, profile.proxy);
   252→      }
   253→    }
   254→
   255→    // Launch URL
   256→    if (profile.launchUrl) {
   257→      args.push(profile.launchUrl);
   258→    }
   259→  }
   260→
   261→  return args;
   262→}

   315→function createFirefoxProxyConfig(profileDir: string, proxy: ProfileData['proxy']): void {
   316→  if (!proxy) return;
   317→
   318→  const prefsPath = path.join(profileDir, 'user.js');
   319→  const existing = fs.existsSync(prefsPath) ? fs.readFileSync(prefsPath, 'utf8') : '';
   320→
   321→  let proxyPrefs = existing + '\n// Proxy Configuration\n';
   322→
   323→  switch (proxy.type) {
   324→    case 'http':
   325→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   326→      proxyPrefs += `user_pref("network.proxy.http", "${proxy.host}");\n`;
   327→      proxyPrefs += `user_pref("network.proxy.http_port", ${proxy.port});\n`;
   328→      proxyPrefs += `user_pref("network.proxy.ssl", "${proxy.host}");\n`;
   329→      proxyPrefs += `user_pref("network.proxy.ssl_port", ${proxy.port});\n`;
   330→      break;
   331→    case 'socks4':
   332→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   333→      proxyPrefs += `user_pref("network.proxy.socks", "${proxy.host}");\n`;
   334→      proxyPrefs += `user_pref("network.proxy.socks_port", ${proxy.port});\n`;
   335→      proxyPrefs += `user_pref("network.proxy.socks_version", 4);\n`;
   336→      break;
   337→    case 'socks5':
   338→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   339→      proxyPrefs += `user_pref("network.proxy.socks", "${proxy.host}");\n`;
   340→      proxyPrefs += `user_pref("network.proxy.socks_port", ${proxy.port});\n`;
   341→      proxyPrefs += `user_pref("network.proxy.socks_version", 5);\n`;
   342→      break;
   343→    case 'https':
   344→      proxyPrefs += `user_pref("network.proxy.type", 1);\n`;
   345→      proxyPrefs += `user_pref("network.proxy.http", "${proxy.host}");\n`;
   346→      proxyPrefs += `user_pref("network.proxy.http_port", ${proxy.port});\n`;
   347→      proxyPrefs += `user_pref("network.proxy.ssl", "${proxy.host}");\n`;
   348→      proxyPrefs += `user_pref("network.proxy.ssl_port", ${proxy.port});\n`;
   349→      break;
   350→  }
   351→
   352→  if (proxy.username) {
   353→    proxyPrefs += `user_pref("network.proxy.share_proxy_settings", true);\n`;
   354→    // Firefox doesn't support proxy auth in prefs directly — need extension
   355→    // We'll handle this via the stealth extension
   356→  }
   357→
   358→  proxyPrefs += `user_pref("network.proxy.no_proxies_on", "localhost, 127.0.0.1");\n`;
   359→
   360→  fs.writeFileSync(prefsPath, proxyPrefs, 'utf8');
   361→}

   367→function createStealthExtension(profile: ProfileData): string | null {
   368→  try {
   369→    const isFirefox = profile.browserType === 'firefox';
   370→    const extDir = path.join(app.getPath('userData'), 'browser-profiles', profile.id, 'joe-stealth-extension');
   371→    fs.mkdirSync(extDir, { recursive: true });
   372→
   373→    if (isFirefox) {
   374→      // Copy Firefox extension files
   375→      const sourceDir = path.join(__dirname, '..', '..', 'main', 'extensions', 'firefox');
   376→      if (!copyExtensionFiles(sourceDir, extDir)) {
   377→        // Fallback: create extension files directly
   378→        createFirefoxExtensionFiles(extDir, profile);
   379→      }
   380→    } else {
   381→      // Copy Chrome extension files
   382→      const sourceDir = path.join(__dirname, '..', '..', 'main', 'extensions', 'chrome');
   383→      if (!copyExtensionFiles(sourceDir, extDir)) {
   384→        // Fallback: create extension files directly
   385→        createChromeExtensionFiles(extDir, profile);
   386→      }
   387→    }
   388→
   389→    // Create fingerprint data file that the extension will read
   390→    createFingerprintDataFile(extDir, profile);
   391→
   392→    return extDir;
   393→  } catch (err) {
   394→    console.error('[RealBrowserLauncher] Failed to create stealth extension:', err);
   395→    return null;
   396→  }
   397→}

   399→function copyExtensionFiles(sourceDir: string, targetDir: string): boolean {
   400→  try {
   401→    if (!fs.existsSync(sourceDir)) return false;
   402→
   403→    const files = fs.readdirSync(sourceDir);
   404→    for (const file of files) {
   405→      const src = path.join(sourceDir, file);
   406→      const dst = path.join(targetDir, file);
   407→      fs.copyFileSync(src, dst);
   408→    }
   409→    return true;
   410→  } catch (err) {
   411→    return false;
   412→  }
   413→}

   499→(function() {

   502→  if (!fp) return;
   503→  if (window.__joeStealthApplied) return;
   504→  window.__joeStealthApplied = true;
   505→  try {
   506→    Object.defineProperty(navigator, 'userAgent', { get: () => fp.userAgent });
   507→    Object.defineProperty(navigator, 'platform', { get: () => fp.platform });
   508→    Object.defineProperty(navigator, 'vendor', { get: () => fp.vendor });
   509→    Object.defineProperty(navigator, 'languages', { get: () => fp.languages });
   510→    Object.defineProperty(navigator, 'hardwareConcurrency', { get: () => fp.hardwareConcurrency });

   514→  } catch(e) {}
   515→  try {
   516→    const sr = fp.screenResolution.split('x');
   517→    const ar = fp.availableScreenResolution.split('x');

   523→    Object.defineProperty(screen, 'pixelDepth', { get: () => fp.colorDepth });
   524→  } catch(e) {}

   639→// PROXY TIMEZONE AUTO-DETECTION
   640→// ============================================================
   641→
   642→async function autoDetectTimezoneFromProxy(profile: ProfileData): Promise<void> {
   643→  if (!profile.proxy) return;
   644→
   645→  try {
   646→    const proxyUrl = `${profile.proxy.type}://${profile.proxy.host}:${profile.proxy.port}`;
   647→    console.log(`[RealBrowserLauncher] Auto-detecting timezone for proxy: ${proxyUrl}`);
   648→
   649→    // Use ip-api.com to detect timezone from proxy IP
   650→    const geoData = await fetchGeoLocation(proxyUrl);
   651→    if (geoData) {
   652→      if (geoData.timezone) {
   653→        profile.fingerprint.timezone = geoData.timezone;
   654→        console.log(`[RealBrowserLauncher] Timezone set to: ${geoData.timezone}`);
   655→      }
   656→      if (geoData.countryCode) {
   657→        // Set language based on country
   658→        const langMap: Record<string, string> = {
   659→          US: 'en-US', GB: 'en-GB', DE: 'de-DE', FR: 'fr-FR', ES: 'es-ES',
   660→          IT: 'it-IT', JP: 'ja-JP', KR: 'ko-KR', CN: 'zh-CN', RU: 'ru-RU',
   661→          BR: 'pt-BR', IN: 'en-IN', AU: 'en-AU', CA: 'en-CA', NL: 'nl-NL',
   662→        };
   663→        if (langMap[geoData.countryCode]) {
   664→          profile.fingerprint.language = langMap[geoData.countryCode];
   665→          profile.fingerprint.languages = [langMap[geoData.countryCode], langMap[geoData.countryCode].split('-')[0]];
   666→        }
   667→      }
   668→    }
   669→  } catch (err) {
   670→    console.warn('[RealBrowserLauncher] Failed to auto-detect timezone from proxy:', err);
   671→    // Don't fail the launch — just use the default timezone
   672→  }
   673→}
   674→
   675→function fetchGeoLocation(proxyUrl: string): Promise<any> {
   676→  return new Promise((resolve) => {
   677→    const url = 'http://ip-api.com/json/?fields=status,countryCode,timezone';
   678→
   679→    const req = http.get(url, { timeout: 5000 }, (res) => {
   680→      let data = '';
   681→      res.on('data', (chunk) => data += chunk);
   682→      res.on('end', () => {
   683→        try {
   684→          const parsed = JSON.parse(data);
   685→          if (parsed.status === 'success') {
   686→            resolve(parsed);
   687→          } else {
   688→            resolve(null);
   689→          }
   690→        } catch (e) {
   691→          resolve(null);
   692→        }
   693→      });
   694→    });
   695→
   696→    req.on('error', () => resolve(null));
   697→    req.on('timeout', () => { req.destroy(); resolve(null); });
   698→  });
   699→}
